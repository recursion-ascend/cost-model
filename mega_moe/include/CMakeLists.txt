cmake_minimum_required(VERSION 3.16)
project(MegaMoeProfile)
if(NOT CMAKE_BUILD_TYPE)
 set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(PROFILE_PYTHON "${Python3_EXECUTABLE}" CACHE FILEPATH "Python interpreter")
set(PROFILE_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/configs/two_card.json5" CACHE FILEPATH "Capture configuration")
set(SOC_VERSION ascend950pr_957c CACHE STRING "Target SoC")
set(ASCEND_CANN_PACKAGE_PATH $ENV{ASCEND_HOME_PATH})
include(${ASCEND_CANN_PACKAGE_PATH}/x86_64-linux/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROFILE_CONFIG}" "${CMAKE_CURRENT_SOURCE_DIR}/include/profile_stages.h" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_profile_events.py" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/config_io.py")
execute_process(COMMAND ${PROFILE_PYTHON} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_profile_events.py"
 --stages "${CMAKE_CURRENT_SOURCE_DIR}/include/profile_stages.h" --config "${PROFILE_CONFIG}" --output "${CMAKE_BINARY_DIR}/generated" RESULT_VARIABLE code)
if(NOT code EQUAL 0)
 message(FATAL_ERROR "Profile event generation failed: ${code}")
endif()
set(SDK_COMMON "${ASCEND_CANN_PACKAGE_PATH}/opp/built-in/op_impl/ai_core/tbe/impl/ops_transformer/ascendc/common")
set(INCS "${REPO_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_BINARY_DIR}/generated"
 "${REPO_ROOT}/mc2/mega_moe/op_kernel/arch35" "${SDK_COMMON}" "${SDK_COMMON}/tensor_api" "${SDK_COMMON}/tensor_api/include"
 "${ASCEND_CANN_PACKAGE_PATH}/include" "${ASCEND_CANN_PACKAGE_PATH}/include/op_common"
 "${ASCEND_CANN_PACKAGE_PATH}/asc/include" "${ASCEND_CANN_PACKAGE_PATH}/asc/include/adv_api"
 "${ASCEND_CANN_PACKAGE_PATH}/asc/include/utils" "${ASCEND_CANN_PACKAGE_PATH}/asc/include/utils/tiling")
ascendc_library(megamoe_kernel SHARED kernel.cpp)
option(ENABLE_PROFILING "Record kernel stages" ON)
set(MEGAMOE_TOPK_PREFETCH 0 CACHE STRING "TopkWeightsPrefetch kernel template switch (0/1)")
set(MEGAMOE_TILE_M 256 CACHE STRING "Scheduler M tile (m-group row height), must match kernel")
set(MEGAMOE_TILE_N 256 CACHE STRING "Scheduler N tile, must match kernel and divide hidden dims")
set(MEGAMOE_L1_BUF_NUM 2 CACHE STRING "BlockMmad L1 ping-pong buffer count (1 disables ping-pong)")
set(MEGAMOE_TOPO_URMA 0 CACHE STRING "Communication topology: 0=MTE window, 1=URMA")
ascendc_compile_definitions(megamoe_kernel PRIVATE MEGAMOE_PROFILE_ABI=1)
ascendc_compile_definitions(megamoe_kernel PRIVATE MEGAMOE_TOPK_PREFETCH=${MEGAMOE_TOPK_PREFETCH})
ascendc_compile_definitions(megamoe_kernel PRIVATE MEGAMOE_TILE_M=${MEGAMOE_TILE_M})
ascendc_compile_definitions(megamoe_kernel PRIVATE MEGAMOE_TILE_N=${MEGAMOE_TILE_N})
ascendc_compile_definitions(megamoe_kernel PRIVATE MEGAMOE_L1_BUF_NUM=${MEGAMOE_L1_BUF_NUM})
if(ENABLE_PROFILING)
 ascendc_compile_definitions(megamoe_kernel PRIVATE ENABLE_PROFILING=1)
endif()
ascendc_include_directories(megamoe_kernel PRIVATE ${INCS})
add_library(megamoe_host SHARED host.cpp)
target_compile_definitions(megamoe_host PRIVATE MEGAMOE_PROFILE_ABI=1 MEGAMOE_TILE_N=${MEGAMOE_TILE_N} MEGAMOE_TOPO_URMA=${MEGAMOE_TOPO_URMA})
target_compile_options(megamoe_host PRIVATE -O2 -std=c++17 -D_GLIBCXX_USE_CXX11_ABI=0)
target_include_directories(megamoe_host PRIVATE ${INCS})
target_link_directories(megamoe_host PRIVATE ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(megamoe_host PRIVATE tiling_api platform)
